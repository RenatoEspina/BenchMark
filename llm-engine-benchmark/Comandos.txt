# JLAMA SERVER — CPU — PUERTO 8080 — TERMINALES 1: servidor, 2: monitor, 3: benchmark

# TERMINAL 1
cd (git rev-parse --show-toplevel)
set -gx JLAMA_IMAGE jlama-docker:latest
docker compose down --remove-orphans >/dev/null 2>&1
or true
set start_ms (date +%s%3N)
docker compose up -d jlama
set started (date +%s)
while not curl -fsS http://127.0.0.1:8080/ui/index.html >/dev/null 2>&1
    if test (math (date +%s) - $started) -ge 900
        docker compose logs --no-color jlama
        exit 1
    end
    sleep 1
end
echo "JLama listo en "(math (date +%s%3N) - $start_ms)" ms"

# TERMINAL 2 — ejecutar después de que la TERMINAL 1 muestre que JLAMA está listo
cd (git rev-parse --show-toplevel)
fish llm-engine-benchmark/scripts/docker-monitor.fish jlama bench-logs/jlama-run1-docker.csv

# TERMINAL 3 — ejecutar mientras el monitor de la TERMINAL 2 está activo
cd (git rev-parse --show-toplevel)
java -jar llm-engine-benchmark/benchmark-app/target/benchmark-app.jar \
    --engine=JLAMA_SERVER \
    --model=tjake/Llama-3.2-1B-Instruct-JQ4 \
    --prompt="Explica en una frase qué es una estructura de datos." \
    --max-tokens=2048 \
    --context=2048 \
    --temperature=0 \
    --rag=false

# TERMINAL 2 — pulsar Ctrl+C al terminar el benchmark y ejecutar
awk -F, '
function mib(value, number) {
    number = value
    gsub(/[^0-9.]/, "", number)
    if (value ~ /GiB/) return number * 1024
    if (value ~ /KiB/) return number / 1024
    return number
}
NR > 1 {
    cpu = $2
    usage = $3
    gsub(/[^0-9.]/, "", cpu)
    split(usage, parts, " / ")
    used = mib(parts[1])
    if (cpu != "") {
        cpu_sum += cpu
        cpu_count++
    }
    if (used > 0) {
        ram_sum += used
        ram_count++
        if (used > peak_ram) peak_ram = used
    }
}
END {
    if (cpu_count > 0) printf "CPU promedio: %.2f %%\n", cpu_sum / cpu_count
    if (ram_count > 0) {
        printf "RAM promedio: %.1f MiB\n", ram_sum / ram_count
        printf "RAM pico: %.1f MiB\n", peak_ram
    }
}
' bench-logs/jlama-run1-docker.csv

# TERMINAL 1 — ejecutar después de obtener los resultados
cd (git rev-parse --show-toplevel)
docker compose down --remove-orphans
