# ============================================================================
# 3. FUNCIONES DE APOYO PARA MEDIR CADA EJECUCIÓN
# ============================================================================

wait_http() {
    local service="$1"
    local url="$2"
    local timeout_seconds="${3:-900}"
    local deadline=$((SECONDS + timeout_seconds))

    until curl -fsS "$url" >/dev/null; do
        if (( SECONDS >= deadline )); then
            echo "Timeout esperando $service en $url"
            docker compose logs --no-color "$service"
            return 1
        fi
        sleep 0.2
    done
}

start_docker_monitor() {
    local container="$1"
    local log="$2"
    printf 'timestamp,cpu_perc,mem_usage,mem_perc\n' > "$log"

    (
        while docker inspect --format '{{.State.Running}}' "$container" 2>/dev/null \
                | grep -q '^true$'; do
            printf '%s,' "$(date --iso-8601=seconds)" >> "$log"
            docker stats "$container" --no-stream \
                --format '{{.CPUPerc}},{{.MemUsage}},{{.MemPerc}}' >> "$log"
            sleep 1
        done
    ) &
    STATS_PID=$!
}

start_gpu_monitor() {
    local log="$1"
    nvidia-smi \
        --query-gpu=timestamp,utilization.gpu,utilization.memory,memory.used,memory.total \
        --format=csv -lms 200 > "$log" 2>&1 &
    GPU_PID=$!
}

stop_monitors() {
    if [ -n "${STATS_PID:-}" ]; then
        kill "$STATS_PID" 2>/dev/null || true
        wait "$STATS_PID" 2>/dev/null || true
        STATS_PID=""
    fi
    if [ -n "${GPU_PID:-}" ]; then
        kill "$GPU_PID" 2>/dev/null || true
        wait "$GPU_PID" 2>/dev/null || true
        GPU_PID=""
    fi
}

run_benchmark() {
    local label="$1"
    local container="$2"
    local engine="$3"
    local model="$4"
    local gpu="$5"

    start_docker_monitor "$container" "$LOG_DIR/$label-docker.csv"
    GPU_PID=""
    if [ "$gpu" = "true" ]; then
        start_gpu_monitor "$LOG_DIR/$label-gpu.csv"
    fi

    set +e
    java -jar "$JAR" \
        "--engine=$engine" \
        "--model=$model" \
        "--prompt=$PROMPT" \
        --max-tokens=128 \
        --context=2048 \
        --temperature=0 \
        --rag=false
    local status=$?
    set -e

    stop_monitors
    return "$status"
}

stop_all() {
    stop_monitors
    docker compose down --remove-orphans >/dev/null 2>&1 || true
}

# Si una request falla o se pulsa Ctrl+C, detiene los monitores y contenedores.
trap stop_all EXIT INT TERM

# ============================================================================

# JLama no usa CUDA; se mide con docker stats, no con nvidia-smi.

    jlama-run1 \
    jlama \
    JLAMA_SERVER \
    tjake/Llama-3.2-1B-Instruct-JQ4 \
    false

# NOTAS DE INTERPRETACIÓN

# - El tiempo "listo" de JLama/vLLM mide el arranque y carga del modelo porque
#   sus servidores no responden hasta terminar esa fase.
# - Ollama arranca primero y descarga el modelo aparte; su load_duration aparece
#   en la respuesta de la API y el benchmark lo imprime como "Carga".
# - JLama 0.8.4 no devuelve usage.completion_tokens y su API ignora temperature;
#   el runner estima tokens por palabras y conserva temperature en la request
#   por compatibilidad.
# - Los tres usan el mismo modelo base Llama 3.2 1B, pero no el mismo formato ni
#   cuantización: JLama usa JQ4, Ollama usa su variante empaquetada y vLLM usa
#   SafeTensors. Eso debe quedar indicado al interpretar los resultados.
