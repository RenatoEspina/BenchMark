# JLAMA SERVER — CPU — PUERTO 8080

# TERMINAL 1 — SERVIDOR + WARM-UP

cd (git rev-parse --show-toplevel)
set -gx JLAMA_IMAGE jlama-docker:latest
docker compose down --remove-orphans >/dev/null 2>&1
or true
set start_ms (date +%s%3N)
docker compose up -d jlama
or begin
    docker compose logs --no-color jlama
    exit 1
end
set started (date +%s)
while not curl -fsS http://127.0.0.1:8080/ui/index.html >/dev/null 2>&1
    set running (docker inspect --format '{{.State.Running}}' jlama 2>/dev/null)
    if test "$running" != "true"
        docker compose logs --no-color --tail=100 jlama
        exit 1
    end
    if test (math (date +%s) - $started) -ge 900
        docker compose logs --no-color --tail=100 jlama
        exit 1
    end
    sleep 1
end
echo "JLama listo en "(math (date +%s%3N) - $start_ms)" ms"

java -jar llm-engine-benchmark/benchmark-app/target/benchmark-app.jar \
    --engine=JLAMA_SERVER \
    --model=tjake/Llama-3.2-1B-Instruct-JQ4 \
    --prompt="Explica en una frase qué es una estructura de datos." \
    --max-tokens=128 \
    --context=2048 \
    --temperature=0 \
    --rag=false >/dev/null

# TERMINAL 2 — MONITOR CPU/RAM

cd (git rev-parse --show-toplevel)
fish llm-engine-benchmark/scripts/docker-monitor.fish \
    jlama bench-logs/jlama-run1-docker.csv

# TERMINAL 3 — BENCHMARK MEDIDO

cd (git rev-parse --show-toplevel)
sleep 5
set run_start_ns (date +%s%N)

java -jar llm-engine-benchmark/benchmark-app/target/benchmark-app.jar \
    --engine=JLAMA_SERVER \
    --model=tjake/Llama-3.2-1B-Instruct-JQ4 \
    --prompt="Explica en una frase qué es una estructura de datos." \
    --max-tokens=128 \
    --context=2048 \
    --temperature=0 \
    --rag=false

set benchmark_status $status
set run_end_ns (date +%s%N)
printf 'start_ns,end_ns\n%s,%s\n' $run_start_ns $run_end_ns \
    > bench-logs/jlama-run1-window.csv

if test $benchmark_status -ne 0
    exit $benchmark_status
end

# TERMINAL 2 — CTRL+C Y RESUMEN

cd (git rev-parse --show-toplevel)
python llm-engine-benchmark/scripts/benchmark-summary.py \
    --window bench-logs/jlama-run1-window.csv \
    --docker bench-logs/jlama-run1-docker.csv \
    --baseline-seconds 5 \
    --activity-floor 5 \
    --activity-margin 5

    echo "La imagen no está descargada."

    docker pull "$VLLM_IMAGE"
    or exit 1
    echo "Imagen descargada."
else
    echo "Imagen Docker encontrada localmente."
end

echo "Deteniendo contenedores anteriores..."
docker compose down --remove-orphans >/dev/null 2>&1
or true

echo "Iniciando vLLM..."
set start_ms (date +%s%3N)

or begin
    echo "Error iniciando el contenedor."
    docker compose logs --no-color vllm
    exit 1
end

echo "Contenedor iniciado."
echo "Mostrando logs de vLLM:"
echo

docker compose logs -f --tail=20 vllm &
set logs_pid $last_pid

set started (date +%s)

while not curl -fsS http://127.0.0.1:8000/health >/dev/null 2>&1
    set running (docker inspect --format '{{.State.Running}}' vllm 2>/dev/null)

    if test "$running" != "true"
        kill $logs_pid >/dev/null 2>&1
        echo
        echo "vLLM terminó antes de estar listo."
        echo "Últimos logs:"
        docker compose logs --no-color --tail=100 vllm
        exit 1
    end

    set elapsed (math (date +%s) - $started)

    if test $elapsed -ge 900
        kill $logs_pid >/dev/null 2>&1
        echo
        echo "Timeout esperando vLLM."
        docker compose logs --no-color --tail=100 vllm
        exit 1
    end

    sleep 1
end

kill $logs_pid >/dev/null 2>&1

echo
echo "vLLM respondió correctamente en /health."
echo "vLLM listo en "(math (date +%s%3N) - $start_ms)" ms"

echo
echo "Ejecutando warm-up..."

java -jar llm-engine-benchmark/benchmark-app/target/benchmark-app.jar \
    --engine=VLLM \
    --model=meta-llama/Llama-3.2-1B-Instruct \
    --prompt="Explica en una frase qué es una estructura de datos." \
    --max-tokens=128 \
    --context=2048 \
    --temperature=0 \
    --rag=false >/dev/null
